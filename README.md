# helper-yao

## Клонирование репозитория (в етрминале)

git clone https://github.com/itcube26/helper-yao

cd helper-yao/


## Создание Python окружения

sudo apt install python3.10-venv

python3 -m venv venv

## Активация окружения

cd helper-yao/

source venv/bin/activate
